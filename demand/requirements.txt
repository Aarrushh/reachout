# demand/ — the demand-dashboard service's own dependencies.
#
# This service is a separate workspace from reachout/ (decision D2: its own
# FastAPI app, its own uvicorn process, its own Postgres schema), so it
# declares its own dependencies rather than borrowing reachout/requirements.txt.
# Everything here is already required by reachout/ too, with one exception,
# noted below.
#
# Python 3.10+ is required to run the full repo test suite from the root:
# reachout/api/ uses `X | None` annotations at module scope, which 3.9 cannot
# import. demand/ itself is 3.9-clean, but do not rely on that.

# Schema validation — the contract gate. Every response and every persisted
# row is validated against demand/shared/schemas/ before it crosses a
# boundary. These schemas are additionalProperties:false and DO NOT MODIFY.
jsonschema>=4.0

# NOT optional, despite looking like it. jsonschema's FormatChecker only
# enforces `format: date-time` when this package is importable — without it,
# "NOPE" passes as a timestamp silently. demand/shared/validation.py computes
# UNENFORCED_FORMATS at import time and it must come out EMPTY; if this dep
# goes missing, every date-time in every schema stops being checked and
# nothing fails loudly to tell you. `format: uuid` and `format: date` are
# enforced by jsonschema itself and do not need this.
rfc3339-validator>=0.1.4

# The demand API app (demand/api/app.py) and its test client.
fastapi>=0.110
uvicorn>=0.29
httpx>=0.27

# PostgREST client. Data ops only — DDL is applied by hand from
# demand/data/schema.sql via the Supabase SQL editor (the anon key cannot
# issue DDL). Note the client is bound to schema "demand"; the shared
# `products` and `stores` tables live in "public" and must be read with an
# explicit .schema("public").
supabase>=2.15

# Loads reachout/.env — one credentials file for the whole repo, the same one
# reachout/api/supa.py reads. Loaded at import time in demand/api/app.py, which
# is where the workspace's only Supabase client is built. (This comment used to
# say "repo-root .env"; nothing loaded any .env at all, so a live run needed the
# vars exported by hand.) Jules VMs hold no keys, so every test must pass with
# fakes and fixtures and never touch this — a missing .env is a no-op.
python-dotenv>=1.0

# Optional: only imported when DEMAND_INGEST_CRON=1 turns on the in-process
# ingest schedule in demand/scripts/run_ingest.py. Same in-process, no-broker
# pattern as the v1 pipeline's simulator.
apscheduler>=3.10

# Live-provider only: imported inside TrendspyProvider in
# demand/ingest/trends_client.py, which is reached solely by
# `run_ingest.py --provider trendspy` (the argparse default). Nothing else needs
# them — `--provider fixture` reads committed JSON, and the whole test suite runs
# on fixtures, so a VM without these still goes green. They are declared here
# because the code imports them: previously neither appeared in any requirements
# file, so the one command V1 actually runs died on ImportError after it had
# already read the database.
trendspy>=0.1
pandas>=2.0

# Dev/test only. Run from the REPO ROOT, never from inside a tests/ directory:
# the suites import demand.* / reachout.*, which resolve only with the repo
# root on sys.path.
pytest>=8.0
